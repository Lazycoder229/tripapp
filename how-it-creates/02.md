```php
<?php

declare(strict_types=1);

namespace Framework\Routing\Attribute;

use Attribute;

/**
 * RouteAttribute class
 * 
 * This class is used to define a route for a controller method.
 * It is an attribute that can be applied to a method to specify the route path and HTTP method.
 * 
 * @package Framework\Routing\Attribute
 */
#[Attribute(Attribute::TARGET_METHOD)]
class Route
{
    public function __construct(
        private string $path,
        private string $method = 'GET'
    ) {
    }

    /**
     * Get the path of the route.
     *
     * @return string The path of the route.
     */
    public function getPath(): string
    {
        return $this->path;
    }
    /**
     * Get the HTTP method of the route.
     *
     * @return string The HTTP method of the route.
     */
    public function getMethod(): string
    {
        return $this->method;
    }
}
```

```php
<?php

declare(strict_types=1);

namespace Framework\Routing;

use Framework\Routing\Attribute\Route;
use ReflectionClass;
use Exception;

/**
 * Router class
 * 
 * Handles registering routes via attributes or manual HTTP methods,
 * matching incoming browser requests, and dispatching them.
 * 
 * @package Framework\Routing
 */
final class Router
{
    public function __construct(
        private array $routes = []
    ) {
    }   

    /**
     * Automatically registers all controller methods that have the Route attribute.
     * 
     * @param string $controllerClass The fully qualified class name of the controller.
     * @return self Returns the Router instance for method chaining.
     */
    public function registerController(string $controllerClass): self
    {
        $reflection = new ReflectionClass($controllerClass);

        foreach ($reflection->getMethods() as $method) {
            $attributes = $method->getAttributes(Route::class);

            foreach ($attributes as $attribute) {
                /** @var Route $routeAttribute */
                $routeAttribute = $attribute->newInstance();

                $path = $routeAttribute->getPath();
                $httpMethod = strtoupper($routeAttribute->getMethod());

                $this->addRoute($httpMethod, $path, [$controllerClass, $method->getName()]);
            }
        }

        return $this;
    }

    /**
     * Registers a route with the specified HTTP method, path, and handler.
     * 
     * @param string $method The HTTP method.
     * @param string $path The route path.
     * @param array|callable $handler The handler for the route.
     * @return void
     */
    private function addRoute(string $method, string $path, array|callable $handler): void
    {
        //Clean the path to ensure it starts with a single slash and has no trailing slashes
        $cleanPath = '/' . trim($path, '/');
        
        // Change the path to a regex pattern for dynamic parameters (e.g., /users/{id} becomes /users/([a-zA-Z0-9_\-]+))
        $regexPattern = preg_replace('/\{[a-zA-Z0-9_]+\}/', '([a-zA-Z0-9_\-]+)', $cleanPath);
        $regexPattern = '#^' . $regexPattern . '$#';

        $this->routes[] = [
            'method' => strtoupper($method),
            'path' => $cleanPath,
            'regex' => $regexPattern,
            'handler' => $handler
        ];
    }

    /**
     * Facilitates the registration of routes for specific HTTP methods.
     * The five methods below (get, post, put, patch, delete) are convenience methods that internally call addRoute with the appropriate HTTP method.
     */
    public function get(string $path, array|callable $handler): void { $this->addRoute('GET', $path, $handler); }
    public function post(string $path, array|callable $handler): void { $this->addRoute('POST', $path, $handler); }
    public function put(string $path, array|callable $handler): void { $this->addRoute('PUT', $path, $handler); }
    public function patch(string $path, array|callable $handler): void { $this->addRoute('PATCH', $path, $handler); }
    public function delete(string $path, array|callable $handler): void { $this->addRoute('DELETE', $path, $handler); }

    /**
     * Reads the current request from the browser and executes the appropriate handler
     */
    public function dispatch(string $requestUri, string $requestMethod): mixed
    {
        // Remove query parameters from the request URI to get the path only
        $parsedUrl = parse_url($requestUri, PHP_URL_PATH);
        $path = '/' . trim($parsedUrl, '/');
        $method = strtoupper($requestMethod);

        foreach ($this->routes as $route) {
            // 1. Match the request method and path against the registered routes
            if ($route['method'] === $method && preg_match($route['regex'], $path, $matches)) {
                
                // Get the parameters from the matched route (if any)
                array_shift($matches); // Remove the full match from the matches array, leaving only the captured groups
                $parameters = $matches;

                $handler = $route['handler'];

                // 2. if the handler is a Closure/Callable (e.g. function() {})
                if (is_callable($handler)) {
                    return call_user_func_array($handler, $parameters);
                }

                // 3. if the handler is an array representing a controller method (e.g. [UserController::class, 'show'])
                if (is_array($handler)) {
                    [$controllerClass, $methodName] = $handler;
                    
                    // Instantiate Controller class
                    $controllerInstance = new $controllerClass();
                        
                        // Execute the method with the parameters and return the result
                    return call_user_func_array([$controllerInstance, $methodName], $parameters);
                }
            }
        }

        // 4. If no matching route is found, throw a 404 Error
        http_response_code(404);
        throw new Exception("404 Not Found: Walang nakitang ruta para sa {$method} {$path}");
    }

    /**
     * Returns all registered routes.
     *
     * @return array An array of all registered routes.
     */
    public function getRoutes(): array
    {
        return $this->routes;
    }
}
```

## Update the Application.php
```php
<?php

declare(strict_types=1);

namespace Framework;

use Framework\Routing\Router;
use RecursiveDirectoryIterator;
use RecursiveIteratorIterator;
use Exception;
/** 
 * Application class
 * 
 * This class is the main entry point for the application.
 * It handles the initialization and execution of the application.
 * 
 * @package Framework
 */
final class Application
{
    private function __construct(){}

    /**
     * Automatically discovers and registers all Controller classes in a given directory
     * 
     */
  private static function autoDiscoverControllers(string $directory, string $namespacePrefix, Router $router): void
    {
        // Check if the directory exists
        if (!is_dir($directory)) {
            return;
        }
        // Use RecursiveDirectoryIterator to traverse the directory and its subdirectories
        $directoryIterator = new RecursiveDirectoryIterator($directory);
        $iterator = new RecursiveIteratorIterator($directoryIterator);

        foreach ($iterator as $file) {
            if ($file->isFile() && $file->getExtension() === 'php') {
                
                // 1. Get filename without .php extension (hal. "HomeController")
                $className = $file->getBasename('.php');

                // 2. Get sub-directory path if there are folders within Controllers (hal. "\Admin" or "")
                $subPath = str_replace(
                    [realpath($directory), $file->getBasename('.php'), '.php'], 
                    ['', '', ''], 
                    $file->getRealPath()
                );

                // Clean the slashes to ensure proper PHP standard namespace format
                $subNamespace = str_replace(['/', '\\'], '\\', $subPath);
                $subNamespace = trim($subNamespace, '\\');

                // 3. Build the full Class Name using the Namespace
                if ($subNamespace !== '') {
                    $controllerClass = $namespacePrefix . '\\' . $subNamespace . '\\' . $className;
                } else {
                    $controllerClass = $namespacePrefix . '\\' . $className;
                }

                // 4. Register with the router if the class actually exists
                if (class_exists($controllerClass)) {
                    $router->registerController($controllerClass);
                }
            }
        }
    }

    /**
     * Runs the application by discovering controllers and dispatching the request.
     * @param string $controllersPath The path to the controllers directory.
     * @param string $controllersNamespace The base namespace for the controllers.
     */
    public static function run(string $controllersPath, string $controllersNamespace): void
    {
        // Create a new Router instance
        $router = new Router();

        // 1. Auto discover and load all controllers from the folder
        self::autoDiscoverControllers($controllersPath, $controllersNamespace, $router);

        try {
            $requestUri = $_SERVER['REQUEST_URI'] ?? '/';
            $requestMethod = $_SERVER['REQUEST_METHOD'] ?? 'GET';

            $response = $router->dispatch($requestUri, $requestMethod);

            if (is_string($response)) {
                echo $response;
            } elseif (is_array($response) || is_object($response)) {
                header('Content-Type: application/json');
                echo json_encode($response);
            }

        } catch (Exception $e) {
            echo "<h1>Framework Error</h1><p>" . $e->getMessage() . "</p>";
        }
    }
}
```

# Put this inside the Controller.php

```php
<?php

declare(strict_types=1);

namespace App\Controller;

use Framework\Routing\Attribute\Route;

class HomeController
{
    /**
     * GET /
     * Para sa pag-load ng pangunahing pahina.
     */
    #[Route('/', 'GET')]
    public function index(): string
    {
        return "Welcome to the home page";
    }

    /**
     * POST /users
     * Karaniwang ginagamit sa paggawa ng bagong resource (e.g., pag-register ng user).
     */
    #[Route('/users', 'POST')]
    public function store(): string
    {
        return "HTTP POST: Successfully created a new user record.";
    }

    /**
     * PUT /users/{id}
     * Ginagamit para sa buong pag-update ng isang umiiral na resource gamit ang ID.
     */
    #[Route('/users/{id}', 'PUT')]
    public function update(string $id): string
    {
        return "HTTP PUT: Fully updated user record with ID: " . $id;
    }

    /**
     * PATCH /users/{id}
     * Ginagamit para sa bahagyang pagbabago (partial update) sa isang resource.
     */
    #[Route('/users/{id}', 'PATCH')]
    public function modify(string $id): string
    {
        return "HTTP PATCH: Partially modified user record with ID: " . $id;
    }

    /**
     * DELETE /users/{id}
     * Ginagamit para sa pagbura ng isang resource mula sa database gamit ang ID.
     */
    #[Route('/users/{id}', 'DELETE')]
    public function destroy(string $id): string
    {
        return "HTTP DELETE: Successfully removed user record with ID: " . $id;
    }
}
```
