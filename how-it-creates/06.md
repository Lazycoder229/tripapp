```php
namespace Framework\Routing\Attribute;

use Attribute;

#[Attribute(Attribute::TARGET_METHOD)]
class Route
{
    public function __construct(
        private string $path,
        private string $method = 'GET',
        private array $middleware = [] // 👈 IDAGDAG ITO: Imbakan ng mga middlewares para sa rutang ito
    ) {
    }

    public function getPath(): string
    {
        return $this->path;
    }

    public function getMethod(): string
    {
        return $this->method;
    }

    public function getMiddleware(): array // 👈 IDAGDAG ITO: Getter para makuha ang mga middleware
    {
        return $this->middleware;
    }
}
```

```php

    public function registerController(string $controllerClass): self
    {
        $reflection = new \ReflectionClass($controllerClass);

        foreach ($reflection->getMethods() as $method) {
            $attributes = $method->getAttributes(Route::class);

            foreach ($attributes as $attribute) {
                /** @var Route $routeInstance */
                $routeInstance = $attribute->newInstance();

                $path = $routeInstance->getPath();
                $httpMethod = strtoupper($routeInstance->getMethod());
                $middleware = $routeInstance->getMiddleware(); // 👈 Kunin ang middleware mula sa attribute

                // Ipasok ang middleware array sa addRoute
                $this->addRoute($httpMethod, $path, [$controllerClass, $method->getName()], $middleware);
            }
        }

        return $this;
    }

    // Dagdagan ng $middleware parameter sa dulo
    private function addRoute(string $method, string $path, array|callable $handler, array $middleware = []): void
    {
        $cleanPath = '/' . trim($path, '/');
        $regexPattern = preg_replace('/\{[a-zA-Z0-9_]+\}/', '([a-zA-Z0-9_\-]+)', $cleanPath);
        $regexPattern = '#^' . $regexPattern . '$#';

        $this->routes[] = [
            'method' => strtoupper($method),
            'path' => $cleanPath,
            'regex' => $regexPattern,
            'handler' => $handler,
            'middleware' => $middleware // 👈 I-save ang middleware sa array ng ruta
        ];
    }
    // I-import ang Pipelines sa itaas ng Router.php kung wala pa:
    // use Framework\Http\Middleware\Pipelines;

    public function dispatch(Request $request): Response
    {
        $path = $request->getPath();
        $method = $request->getMethod();

        foreach ($this->routes as $route) {
            if ($route['method'] === $method && preg_match($route['regex'], $path, $matches)) {
                array_shift($matches);
                $parameters = $matches;

                // Gumawa ng callback para sa destinasyon (ang totoong pagtakbo ng Controller)
                $destination = function (Request $request) use ($route, $parameters): Response {
                    $handler = $route['handler'];
                    
                    // Isama ang request sa parameters ng controller gaya ng dati
                    array_unshift($parameters, $request);

                    if (is_callable($handler)) {
                        $response = call_user_func_array($handler, $parameters);
                    } else {
                        [$controllerClass, $methodName] = $handler;
                        $controllerInstance = $this->container->get($controllerClass);
                        $response = call_user_func_array([$controllerInstance, $methodName], $parameters);
                    }

                    if ($response instanceof Response) {
                        return $response;
                    }
                    if (is_array($response) || is_object($response)) {
                        return Response::json($response);
                    }
                    return new Response((string) $response);
                };

                // KUNG MAY MIDDLEWARE ANG RUTA: Patakbuhin ito sa loob ng Pipeline!
                if (!empty($route['middleware'])) {
                    $pipeline = new Pipelines($this->container);
                    $pipeline->pipe($route['middleware']);
                    
                    return $pipeline->process($request, $destination);
                }

                // Kung walang middleware ang partikular na ruta, dumeretso agad sa controller
                return $destination($request);
            }
        }

        return new Response("404 Not Found: No route found for {$method} {$path}", 404);
    }
```

```php
<?php

declare(strict_types=1);

namespace App\Controller;

use Framework\Routing\Attribute\Route;
use Framework\Http\Request;
use App\Middleware\AuthMiddleware; // I-import ang iyong middleware class

class HomeController
{
    /**
     * Libre itong mabisita ng kahit sino! (Walang middleware)
     * URL: http://localhost:3000/
     */
    #[Route('/', 'GET')]
    public function index(Request $request): string
    {
        return "Welcome to the PUBLIC home page! Kahit walang token, pasok ka dito.";
    }

    /**
     * PROTEKTADONG RUTA: Dadaan muna ito sa AuthMiddleware bago makapasok!
     * URL: http://localhost:3000/dashboard?token=secret123
     */
    #[Route('/dashboard', 'GET', [AuthMiddleware::class])] // 👈 IPASA ANG MIDDLEWARE DITO!
    public function dashboard(Request $request): string
    {
        return "Welcome sa iyong SECURE Dashboard! Kasi tama ang iyong token.";
    }
}
```
