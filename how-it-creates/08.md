<?php

declare(strict_types=1);

namespace Framework\Http;

/**
 * This class encapsulates HTTP request data, providing methods to access query parameters, request body data, server information, and headers.
 * It also supports method spoofing for HTML forms and JSON payloads for API requests.
 * @package Framework\Http
 */
final class Request
{
    public function __construct(
        private array $query,
        private array $body,
        private array $server,
        private array $headers = [], // 👈 1. IDINAGDAG DITO: Imbakan ng mga nalinis na request headers
        private ?string $rawContent = null
    ) {
    }

    /**
     * Factory method to boot a Request object up using native PHP global boundaries
     * 
     * @return self Returns a new instance of the Request class populated with global request data.
     */
    public static function createFromGlobals(): self
    {
        // Capture the global request data
        $body = $_POST;
        $server = $_SERVER;
        $method = strtoupper($server['REQUEST_METHOD'] ?? 'GET');

        // Bago mong code para sa ligtas na Method Spoofing
        $allowedSpoofMethods = ['PUT', 'PATCH', 'DELETE'];
        if ($method === 'POST' && isset($_POST['_method'])) {
            $spoofed = strtoupper($_POST['_method']);

            if (in_array($spoofed, $allowedSpoofMethods, true)) {
                $server['REQUEST_METHOD'] = $spoofed;
                $method = $spoofed;
            }
        }

        // Support native JSON API payload streams
        $rawContent = file_get_contents('php://input');
        if ($method !== 'GET' && !empty($rawContent)) {
            $jsonData = json_decode($rawContent, true);
            if (is_array($jsonData)) {
                $body = array_merge($body, $jsonData);
            }
        }

        // 👈 2. IDINAGDAG DITO: Proseso para i-extract at linisin ang headers mula sa $_SERVER
        $headers = [];
        foreach ($server as $key => $value) {
            if (str_starts_with($key, 'HTTP_')) {
                // Pinapalitan ang HTTP_USER_AGENT patungong user-agent
                $name = str_replace('_', '-', strtolower(substr($key, 5)));
                $headers[$name] = $value;
            } elseif (in_array($key, ['CONTENT_TYPE', 'CONTENT_LENGTH'])) {
                $name = str_replace('_', '-', strtolower($key));
                $headers[$name] = $value;
            }
        }

        // Return a new Request instance populated with the captured data kabilang ang $headers
        return new self($_GET, $body, $server, $headers, $rawContent ?: null);
    }

    /**
     * Get the current request path (URI) without query parameters.
     * @return string Returns the request path, e.g., "/users/123".
     */
    public function getPath(): string
    {
        $uri = $this->server['REQUEST_URI'] ?? '/';
        $parsedUrl = parse_url($uri, PHP_URL_PATH);
        return '/' . trim($parsedUrl ?: '/', '/');
    }

    /**
     * Get the HTTP method of the current request (GET, POST, PUT, DELETE, etc.).
     * @return string Returns the HTTP method in uppercase.
     */
    public function getMethod(): string
    {
        return strtoupper($this->server['REQUEST_METHOD'] ?? 'GET');
    }

    /**
     * 👈 3. IDINAGDAG NA PUBLIC API: Kumuha ng partikular na request header value (Case-insensitive)
     * Halimbawa: $request->header('Authorization') o $request->header('X-Requested-With')
     */
    public function header(string $key, ?string $default = null): ?string
    {
        $normalizedKey = str_replace('_', '-', strtolower($key));
        return $this->headers[$normalizedKey] ?? $default;
    }

    /**
     * 👈 4. IDINAGDAG NA PUBLIC API: Kunin ang lahat ng request headers na ipinadala ng browser/client
     */
    public function headers(): array
    {
        return $this->headers;
    }

    /**
     * Fetch from URL Query Params ($_GET)
     */
    public function query(string $key, mixed $default = null): mixed
    {
        return $this->query[$key] ?? $default;
    }

    /**
     * Fetch from Request Body ($_POST or parsed JSON payloads)
     */
    public function input(string $key, mixed $default = null): mixed
    {
        return $this->body[$key] ?? $default;
    }

    /**
     * Merge and dump all input arrays
     */
    public function all(): array
    {
        return array_merge($this->query, $this->body);
    }
}

<?php

declare(strict_types=1);

namespace Framework\Http;

/**
 * This class handles setting HTTP response status codes, bodies, and outbound custom headers safely.
 * @package Framework\Http
 */
class Response
{
    public function __construct(
        protected mixed $content,
        protected int $statusCode = 200,
        protected array $headers = []
    ) {
        // Tinitiyak na ang keys ng paunang headers ay lowercase para sa case-insensitivity
        $normalizedHeaders = [];
        foreach ($headers as $name => $value) {
            $normalizedHeaders[strtolower($name)] = $value;
        }
        $this->headers = $normalizedHeaders;
    }

    /**
     * Named constructor helper for fast structural JSON API outputs
     */
    public static function json(array|object $data, int $statusCode = 200): self
    {
        return new self(
            json_encode($data),
            $statusCode,
            ['Content-Type' => 'application/json']
        );
    }

    /**
     * PUBLIC API: Fluent modifier para magdagdag ng custom output headers
     * Halimbawa: return $response->withHeader('X-Powered-By', 'MyOwnFramework');
     */
    public function withHeader(string $name, string $value): self
    {
        $this->headers[strtolower($name)] = $value;
        return $this;
    }

    /**
     * PUBLIC API: Alisin ang isang nakatakdang response header
     */
    public function withoutHeader(string $name): self
    {
        unset($this->headers[strtolower($name)]);
        return $this;
    }

    public function getHeaders(): array
    {
        return $this->headers;
    }

    /**
     * Itatapon ang status, lahat ng rehistradong headers, at ang mismong content sa browser ng user
     */
    public function send(): void
    {
        http_response_code($this->statusCode);

        foreach ($this->headers as $name => $value) {
            $formattedName = ucwords($name, '-');
            header("{$formattedName}: {$value}");
        }

        echo $this->content;
    }
}

To complete this method using your newly added Headers API, you need to remember that your controller method is type-hinted to return an array, but Response::header doesn't exist as a static property.If you want to return actual header details as an array response, you can extract them using the new $request->headers() method or send them back via a proper Response object wrapper.Here are the 2 best ways to complete your method depending on what you want to achieve:Option A: Return an array of all incoming request headers 📥If your goal is to show the user all the headers their browser or API client sent to the server (like User-Agent, Host, etc.):php  

#[Route('/public', 'GET')]
    public function publicData(Request $request): array
    {
        // This uses the new $request->headers() API method from scratch
        return $request->headers();
    }
    
Use code with caution.What happens: Your framework's router will automatically catch this array, format it as JSON, and print it nicely on the screen.

Option B: Return a response with custom outbound headers 📤If you want to change your method signature from array to Response so you can attachment custom security or custom metadata headers back to the client browser:php    use Framework\Http\Response; // Make sure to import this at the top of your file

    #[Route('/public', 'GET')]
    public function publicData(Request $request): Response
    {
        $data = [
            'status' => 'Success',
            'message' => 'Welcome to the public endpoint!'
        ];

        // 1. Wrap the array inside our scratch Response class
        $response = Response::json($data);

        // 2. Fluidly attach custom headers using your new withHeader API
        return $response
            ->withHeader('X-Framework-Developer', 'You')
            ->withHeader('X-Custom-Security-Key', '987654321')
            ->withHeader('Cache-Control', 'max-age=3600');
    }
Use code with caution.What happens: When you open this in your browser or Postman, you will see the JSON data, and if you check the network tabs, your custom headers will be attached perfectly to the network life cycle.

## update again the router

```php

<?php

declare(strict_types=1);

namespace Framework\Routing;

use Framework\Routing\Attribute\Route;
use Framework\Container\Container;
use Framework\Http\Request;
use Framework\Http\Response;
use Framework\Http\Middleware\Pipelines;
use ReflectionClass;
use Exception;

/**
 * Router class
 * 
 * Handles registering routes via attributes or manual HTTP methods,
 * matching incoming browser requests, and dispatching them.
 * 
 * @package Framework\Routing
 */
final class Router
{
    public function __construct(
        private Container $container,
        private array $routes = []
    ) {
    }

    /**
     * Automatically registers all controller methods that have the Route attribute.
     * Supports class-level Route for prefix and default middleware inheritance.
     * 
     * @param string $controllerClass The fully qualified class name of the controller.
     * @return self Returns the Router instance for method chaining.
     */
    public function registerController(string $controllerClass): self
    {
        $reflection = new \ReflectionClass($controllerClass);

        // --- Class-level Route (optional) ---
        // Kung may #[Route] sa class, kunin ang path prefix at default middleware nito.
        // Ang lahat ng method routes ay magmamana ng prefix at middleware na ito.
        $classAttributes = $reflection->getAttributes(Route::class);
        $classPrefix = '';
        $classMiddleware = [];

        if (!empty($classAttributes)) {
            $classRoute = $classAttributes[0]->newInstance();
            $classPrefix = rtrim($classRoute->getPath(), '/');       // e.g. '/api/v1'
            $classMiddleware = $classRoute->getMiddleware();          // e.g. ['api.secure']
        }

        // --- Method-level Routes ---
        // I-scan ang bawat method ng controller para sa #[Route] attribute.
        foreach ($reflection->getMethods() as $method) {
            $attributes = $method->getAttributes(Route::class);

            foreach ($attributes as $attribute) {
                $routeInstance = $attribute->newInstance();

                // I-combine ang class prefix + method path, at linisin ang slashes
                // e.g. '/api/v1' + '/secure' = '/api/v1/secure'
                $rawPath = '/' . trim($classPrefix, '/') . '/' . trim($routeInstance->getPath(), '/');
                $path = '/' . trim($rawPath, '/');

                $httpMethod = strtoupper($routeInstance->getMethod());

                // I-merge ang class-level middleware + method-level middleware
                // Class middleware ay unang tatakbo, tapos method middleware
                $middleware = array_merge($classMiddleware, $routeInstance->getMiddleware());

                $this->addRoute($httpMethod, $path, [$controllerClass, $method->getName()], $middleware);
            }
        }

        return $this;
    }

    /**
     * Registers a route internally with its method, path, handler, and middleware.
     * Converts path parameters like {id} into regex capture groups.
     */
    private function addRoute(string $method, string $path, array|callable $handler, array $middleware = []): void
    {
        $cleanPath = '/' . trim($path, '/');

        // I-convert ang {param} placeholders patungong regex capture groups
        // e.g. /users/{id} → #^/users/([a-zA-Z0-9_\-]+)$#
        $regexPattern = preg_replace('/\{[a-zA-Z0-9_]+\}/', '([a-zA-Z0-9_\-]+)', $cleanPath);
        $regexPattern = '#^' . $regexPattern . '$#';

        $this->routes[] = [
            'method'     => strtoupper($method),
            'path'       => $cleanPath,
            'regex'      => $regexPattern,
            'handler'    => $handler,
            'middleware' => $middleware,
        ];
    }

    /**
     * Convenience methods for registering routes per HTTP method.
     * Internally calls addRoute with the appropriate HTTP verb.
     */
    public function get(string $path, array|callable $handler): void    { $this->addRoute('GET',    $path, $handler); }
    public function post(string $path, array|callable $handler): void   { $this->addRoute('POST',   $path, $handler); }
    public function put(string $path, array|callable $handler): void    { $this->addRoute('PUT',    $path, $handler); }
    public function patch(string $path, array|callable $handler): void  { $this->addRoute('PATCH',  $path, $handler); }
    public function delete(string $path, array|callable $handler): void { $this->addRoute('DELETE', $path, $handler); }

    /**
     * Matches the incoming request to a registered route and dispatches it.
     * Runs route-level middleware pipeline before reaching the controller.
     */
    public function dispatch(Request $request): Response
    {
        $path   = $request->getPath();
        $method = $request->getMethod();

        foreach ($this->routes as $route) {
            // Suriin kung tugma ang HTTP method at URL path
            if ($route['method'] !== $method || !preg_match($route['regex'], $path, $matches)) {
                continue;
            }

            // Alisin ang full match, itago lang ang captured route parameters
            array_shift($matches);
            $parameters = $matches;

            // Ang destination ay ang pinakahuli — ang controller method mismo
            $destination = function (Request $request) use ($route, $parameters): Response {
                $handler = $route['handler'];

                // Ipasok ang Request bilang unang argument sa controller method
                array_unshift($parameters, $request);

                if (is_callable($handler)) {
                    $response = call_user_func_array($handler, $parameters);
                } else {
                    [$controllerClass, $methodName] = $handler;
                    $controllerInstance = $this->container->get($controllerClass);
                    $response = call_user_func_array([$controllerInstance, $methodName], $parameters);
                }

                // I-normalize ang return value ng controller patungong Response object
                if ($response instanceof Response) {
                    return $response;                      // direktang Response — i-return na
                }
                if (is_array($response) || is_object($response)) {
                    return Response::json($response);      // array/object → JSON response
                }
                return new Response((string) $response);   // string → plain HTML response
            };

            // --- Middleware Resolution ---
            // Kung may middleware ang route, i-resolve muna ang aliases/groups
            // bago patakbuhin ang pipeline.
            if (!empty($route['middleware'])) {
                $resolvedMiddlewares = [];
                $groups = \Framework\Application::getMiddlewareGroups();

                foreach ($route['middleware'] as $item) {
                    if (isset($groups[$item])) {
                        // Ang item ay isang registered alias o group — i-expand ang mga klase nito
                        // e.g. 'api.secure' → [RateLimitMiddleware::class, AuthMiddleware::class]
                        $resolvedMiddlewares = array_merge($resolvedMiddlewares, $groups[$item]);
                    } else {
                        // Ang item ay direktang class name na — gamitin na siya nang direkta
                        $resolvedMiddlewares[] = $item;
                    }
                }

                // Tanggalin ang mga duplicate na klase para hindi mag-double run
                $resolvedMiddlewares = array_unique($resolvedMiddlewares);

                // Patakbuhin ang middleware pipeline bago ang controller
                $pipeline = new Pipelines($this->container);
                $pipeline->pipe($resolvedMiddlewares);

                return $pipeline->process($request, $destination);
            }

            // Kung walang middleware ang route, direkta na sa controller
            return $destination($request);
        }

        // Walang nahanap na route — i-return ang 404
        return new Response("404 Not Found: No route found for {$method} {$path}", 404);
    }

    /**
     * Returns all registered routes.
     * Useful for debugging or route inspection.
     *
     * @return array An array of all registered routes.
     */
    public function getRoutes(): array
    {
        return $this->routes;
    }
}
