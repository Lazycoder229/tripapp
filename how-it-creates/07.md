## Middleware Attribute

```bash
Gumawa ng Middleware.php Attribute classGagawa tayo ng attribute na magtatakda kung ano ang alias (maikling pangalan) o groups na kinabibilangan ng isang middleware class.Gumawa ng file sa src/Framework/Http/Middleware/Attribute/Middleware.php:
```

```php
<?php

declare(strict_types=1);

namespace Framework\Http\Middleware\Attribute;

use Attribute;

#[Attribute(Attribute::TARGET_CLASS)] // Inilalagay ito sa ibabaw ng CLASS, hindi sa method
class Middleware
{
    public function __construct(
        private string $alias,
        private array $groups = []
    ) {
    }

    public function getAlias(): string
    {
        return $this->alias;
    }

    public function getGroups(): array
    {
        return $this->groups;
    }
}
```

```bash
 Idagdag ang Auto-Discovery Logic sa Application.phpMagdaragdag tayo ng autoDiscoverMiddlewares method sa iyong Application.php upang i-scan ang middleware folder gamit ang Reflection API para basahin ang #[Middleware] attribute.Buksan ang iyong src/Framework/Application.php at i-update ang buong file
```
```php
<?php

declare(strict_types=1);

namespace Framework;

use Framework\Routing\Router;
use Framework\Container\Container; 
use Framework\Http\Request;
use Framework\Http\Response;
use Framework\Http\Middleware\Pipelines;
use Framework\Http\Middleware\Attribute\Middleware; // I-import ang bagong attribute
use RecursiveDirectoryIterator;
use RecursiveIteratorIterator;
use Exception;
use ReflectionClass;

final class Application
{
    private static array $middlewareGroups = [];

    private function __construct(){}

    /**
     * Awtomatikong nag-iiscan at nagrerehistro ng Middlewares base sa Attributes
     */
    private static function autoDiscoverMiddlewares(string $directory, string $namespacePrefix): void
    {
        if (!is_dir($directory)) {
            return;
        }

        $iterator = new RecursiveIteratorIterator(new RecursiveDirectoryIterator($directory));
        $basePath = str_replace('\\', '/', realpath($directory));

        foreach ($iterator as $file) {
            if ($file->isFile() && $file->getExtension() === 'php') {
                $className = $file->getBasename('.php');
                $currentFilePath = str_replace('\\', '/', $file->getRealPath());
                $subPath = str_replace([$basePath, '/' . $file->getBasename()], ['', ''], $currentFilePath);
                $subNamespace = trim(str_replace('/', '\\', $subPath), '\\');

                $middlewareClass = $subNamespace !== '' 
                    ? $namespacePrefix . '\\' . $subNamespace . '\\' . $className 
                    : $namespacePrefix . '\\' . $className;

                if (class_exists($middlewareClass)) {
                    $reflection = new ReflectionClass($middlewareClass);
                    $attributes = $reflection->getAttributes(Middleware::class);

                    foreach ($attributes as $attribute) {
                        /** @var Middleware $middlewareAttr */
                        $middlewareAttr = $attribute->newInstance();
                        $alias = $middlewareAttr->getAlias();

                        // 1. I-rehistro ang nag-iisang alias (e.g., ['auth' => [AuthMiddleware::class]])
                        self::$middlewareGroups[$alias] = [$middlewareClass];

                        // 2. I-rehistro sa mga grupo kung may tinakda (e.g., 'api.secure')
                        foreach ($middlewareAttr->getGroups() as $group) {
                            if (!isset(self::$middlewareGroups[$group])) {
                                self::$middlewareGroups[$group] = [];
                            }
                            self::$middlewareGroups[$group][] = $middlewareClass;
                        }
                    }
                }
            }
        }
    }

    private static function autoDiscoverControllers(string $directory, string $namespacePrefix, Router $router): void
    {
        if (!is_dir($directory)) {
            return;
        }

        $iterator = new RecursiveIteratorIterator(new RecursiveDirectoryIterator($directory));
        $basePath = str_replace('\\', '/', realpath($directory));

        foreach ($iterator as $file) {
            if ($file->isFile() && $file->getExtension() === 'php') {
                $className = $file->getBasename('.php');
                $currentFilePath = str_replace('\\', '/', $file->getRealPath());
                $subPath = str_replace([$basePath, '/' . $file->getBasename()], ['', ''], $currentFilePath);
                $subNamespace = trim(str_replace('/', '\\', $subPath), '\\');

                $controllerClass = $subNamespace !== '' 
                    ? $namespacePrefix . '\\' . $subNamespace . '\\' . $className 
                    : $namespacePrefix . '\\' . $className;

                if (class_exists($controllerClass)) {
                    $router->registerController($controllerClass);
                }
            }
        }
    }

    public static function getMiddlewareGroups(): array
    {
        return self::$middlewareGroups;
    }

    /**
     * Ginawang dynamic ang run method para tanggapin din ang middleware configurations
     */
    public static function run(
        string $controllersPath, 
        string $controllersNamespace,
        string $middlewaresPath,       // 👈 IDAGDAG ITO
        string $middlewaresNamespace   // 👈 IDAGDAG ITO
    ): void {
        $container = new Container();
        $router = new Router($container);

        // 1. Unang i-auto discover ang mga middlewares para handa na sila bago mag-load ang router
        self::autoDiscoverMiddlewares($middlewaresPath, $middlewaresNamespace);

        // 2. I-auto discover ang mga controllers
        self::autoDiscoverControllers($controllersPath, $controllersNamespace, $router);

        $request = Request::createFromGlobals();

        try {
            $pipeline = new Pipelines($container);
            
            $response = $pipeline->process($request, function (Request $request) use ($router): Response {
                return $router->dispatch($request);
            });

        } catch (Exception $e) {
            $response = new Response("<h1>Framework Critical Error</h1><p>{$e->getMessage()}</p>", 500);
        }

        $response->send();
    }
}
```

```bash
 Lagyan ng Attribute ang iyong MiddlewaresNgayon, bubuksan natin ang iyong mga middleware classes upang lagyan sila ng #[Middleware] attribute sa pinakataas. Dito natin idedeklara ang kanilang alias at kung anong grupo sila kasama.I-update ang AuthMiddleware.php:
```

```php
<?php

namespace App\Middleware;

use Framework\Http\Middleware\MiddlewareInterface;
use Framework\Http\Middleware\Attribute\Middleware; // I-import ang bagong attribute
use Framework\Http\Request;
use Framework\Http\Response;

// Awtomatikong magiging 'auth' ang alias nito, at kasama rin ito sa grupong 'api.secure'
#[Middleware(alias: 'auth', groups: ['api.secure'])]
class AuthMiddleware implements MiddlewareInterface
{
    public function handle(Request $request, callable $next): Response
    {
        $token = $request->query('token');

        if ($token !== 'secret123') {
            return Response::json(['error' => 'Unauthorized'], 401);
        }

        return $next($request);
    }
}
```

```php
                // KUNG MAY MIDDLEWARE ANG RUTA: I-resolve muna ang mga aliases at groups
                if (!empty($route['middleware'])) {
                    $resolvedMiddlewares = [];
                    
                    // Kunin ang mga auto-discovered groups mula sa Application class
                    $groups = \Framework\Application::getMiddlewareGroups();

                    foreach ($route['middleware'] as $item) {
                        // Case 1: Kung ang nakasulat ay isang rehistradong grupo (hal. 'api.secure')
                        if (isset($groups[$item])) {
                            // Dahil ang grupo ay array ng mga classes, pagsamahin natin sila
                            $resolvedMiddlewares = array_merge($resolvedMiddlewares, $groups[$item]);
                        } 
                        // Case 2: Kung ang nakasulat ay isang single alias na ginawa ng single middleware
                        elseif (isset($groups[$item])) {
                            $resolvedMiddlewares = array_merge($resolvedMiddlewares, $groups[$item]);
                        }
                        // Case 3: Kung walang tugmang alias, ibig sabihin ito ay direktang Class Name na (e.g. AuthMiddleware::class)
                        else {
                            $resolvedMiddlewares[] = $item;
                        }
                    }

                    // Tanggalin ang mga duplicate kung sakaling ma-double register ang isang klase
                    $resolvedMiddlewares = array_unique($resolvedMiddlewares);

                    // Patakbuhin ang Pipelines gamit ang mga na-resolve na totoong klase
                    $pipeline = new Pipelines($this->container);
                    $pipeline->pipe($resolvedMiddlewares);
                    
                    return $pipeline->process($request, $destination);
                }
```
